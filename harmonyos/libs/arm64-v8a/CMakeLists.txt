# ============================================================================
# CMakeLists.txt - native_process NDK 原生库构建配置
#
# 用于在 DevEco Studio 中编译 native_process.c 为 libnative_process.so，
# 供鸿蒙 ArkTS 通过 Node-API (NAPI) 调用。
#
# 构建产物: libnative_process.so (ARM64)
# 依赖库:   libace_napi.z.so (NAPI), liblog.z.so (鸿蒙日志)
# ============================================================================

cmake_minimum_required(VERSION 3.5.0)
project(native_process)

# ----------------------------------------------------------------------------
# C 标准设置（C99）
# ----------------------------------------------------------------------------
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)

# ----------------------------------------------------------------------------
# 头文件包含目录
# ----------------------------------------------------------------------------
include_directories(${CMAKE_CURRENT_SOURCE_DIR})

# ----------------------------------------------------------------------------
# 构建共享库
# ----------------------------------------------------------------------------
add_library(native_process SHARED
    native_process.c
)

# ----------------------------------------------------------------------------
# 链接鸿蒙系统库
#   libace_napi.z.so - Node-API 接口（ArkTS ↔ C 交互）
#   liblog.z.so      - 鸿蒙 hilog 日志系统
# ----------------------------------------------------------------------------
target_link_libraries(native_process
    libace_napi.z.so
    liblog.z.so
)

# ----------------------------------------------------------------------------
# 编译选项
#   -fPIC  位置无关代码（共享库必需）
#   -O2    优化级别
#   -Wall  启用所有警告
# ----------------------------------------------------------------------------
target_compile_options(native_process PRIVATE
    -fPIC
    -O2
    -Wall
)

# ----------------------------------------------------------------------------
# 输出库名（CMake 会自动加 lib 前缀和 .so 后缀）
# ----------------------------------------------------------------------------
set_target_properties(native_process PROPERTIES
    OUTPUT_NAME "native_process"
)
